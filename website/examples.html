Examples

Slint UI definitions and Rust integrations

Running the Examples

All examples are in the examples/ directory. Each .slint file defines a reusable UI component. The Rust code loads it via Helium::from_file().

Example structure:

examples/
  minimal-bar.slint     # Simple centered text
  clock-bar.slint       # Right-aligned clock (Space Mono)
  workspace-bar.slint   # Workspace indicators
  full-bar.slint        # Combined layout (imports above)
  README.md

Minimal Bar

A bare-bones bar with a single centered text label.

// examples/minimal-bar.slint
export component MinimalBar {
    width: 1920px;
    height: 42px;

    in property <string> label;

    Rectangle {
        background: #141414;
        Text {
            text: label;
            color: #d4d4d4;
            font-size: 13px;
            font-family: "Space Grotesk";
            vertical-alignment: center;
            horizontal-alignment: center;
        }
    }
}
use helium_wsl::{AnchorEdge, Helium};

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let mut runtime = Helium::from_file("examples/minimal-bar.slint")
        .surface("Main")      // matches component name
        .size(1920, 42)
        .anchor((AnchorEdge::Top, AnchorEdge::Left, AnchorEdge::Right))
        .exclusive()
        .build()?;

    runtime.run()?;
    Ok(())
}

Clock Bar

A right-aligned clock display using Space Mono for tabular numerals.

// examples/clock-bar.slint
export component ClockBar {
    width: 1920px;
    height: 42px;

    in property <string> clock_text;

    Rectangle {
        background: #141414;
        Text {
            text: clock_text;
            color: #76b900;
            font-size: 14px;
            font-family: "Space Mono";
            letter-spacing: 0.1em;
            vertical-alignment: center;
            horizontal-alignment: right;
            padding-right: 16px;
        }
    }
}
use helium_wsl::{
    AnchorEdge, Helium,
    adapters::{adapters, ClockAdapter, AdapterRegistry, AdapterCtx},
};

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let mut runtime = Helium::from_file("examples/clock-bar.slint")
        .surface("ClockBar")
        .size(200, 42)
        .anchor((AnchorEdge::Top, AnchorEdge::Right))
        .build()?;

    // The ClockAdapter sets clock_text on each tick
    let mut ctx = AdapterCtx::_new();
    let mut registry = AdapterRegistry::new(adapters! {
        "clock" => ClockAdapter { format: "%H:%M:%S".into(), ..Default::default() },
    });
    registry.init_all(&mut ctx);

    runtime.run()?;
    Ok(())
}

Workspace Bar

Workspace indicators with active-state highlighting.

// examples/workspace-bar.slint
export component WorkspaceBar {
    width: 1920px;
    height: 42px;

    in property <int> active_workspace;
    in property <string> workspace_1;
    in property <string> workspace_2;
    in property <string> workspace_3;
    in property <string> workspace_4;
    in property <string> workspace_5;

    Rectangle {
        background: #141414;
        HorizontalLayout {
            padding: 8px 12px;
            spacing: 4px;

            Rectangle {
                width: 28px; height: 26px;
                background: active_workspace == 1 ? #76b900 : #1f1f1f;
                border: 1px solid active_workspace == 1 ? #76b900 : #333;
                Text {
                    text: workspace_1;
                    color: active_workspace == 1 ? #0a0a0a : #888;
                    font-size: 11px;
                    font-family: "Space Mono";
                    vertical-alignment: center;
                    horizontal-alignment: center;
                }
            }
            // ... repeats for workspaces 2-5
        }
    }
}
use helium_wsl::{
    AnchorEdge, Helium,
    adapters::{WorkspacesAdapter},
};

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let mut runtime = Helium::from_file("examples/workspace-bar.slint")
        .surface("WorkspaceBar")
        .size(180, 42)
        .anchor((AnchorEdge::Top, AnchorEdge::Left))
        .build()?;

    runtime.run()?;
    Ok(())
}

Full Bar

A complete bar combining workspaces, a center label, and a clock — using Slint component imports.

// examples/full-bar.slint
import { MinimalBar } from "./minimal-bar.slint";
import { ClockBar } from "./clock-bar.slint";
import { WorkspaceBar } from "./workspace-bar.slint";

export component FullBar {
    width: 1920px;
    height: 42px;

    in property <string> label;
    in property <string> clock_text;
    in property <int> active_workspace;
    in property <string> workspace_1;
    in property <string> workspace_2;
    in property <string> workspace_3;
    in property <string> workspace_4;
    in property <string> workspace_5;

    Rectangle {
        background: #0d0d0d;
        HorizontalLayout {
            WorkspaceBar {
                width: 180px; height: 42px;
                active_workspace: active_workspace;
                workspace_1: workspace_1;
                // ...
            }
            MinimalBar {
                width: 1200px; height: 42px;
                label: label;
            }
            ClockBar {
                width: 200px; height: 42px;
                clock_text: clock_text;
            }
        }
    }
}
